amenities_data <- merge(urban_data, parks_data, by = "NeighborhoodGroup") %>% 
  rename(
    `UCC` = `Urban Center Count`,
    `PC` = `Park Count`
  ) %>% 
  mutate(`Total` = `UCC` + `PC`)

amenities_data <- merge(amenities_data, population_data, by = "NeighborhoodGroup") %>% 
  select(NeighborhoodGroup, `Total Population`, UCC, PC, Total) %>% 
  group_by(NeighborhoodGroup) %>% 
  summarize(`Population` = sum(`Total Population`),
           UCC = sum(UCC, na.rm = TRUE) / n(),
    PC = sum(PC, na.rm = TRUE) / n(),
    Total = sum(Total, na.rm = TRUE) / n())


amenities_data <- amenities_data %>% 
  mutate(`Amenities Rate` = (Total / Population) * 10000)

ggplot(amenities_data, aes(x = NeighborhoodGroup, y = `Amenities Rate`, fill = factor(NeighborhoodGroup))) +
  coord_flip() +
  geom_col() +
  labs(
    title = "Amenities Distribution by Neighborhood",
    x = "Neighborhood",
    y = "Number of Amenities per 10,000 Residents",
    fill = "Neighborhood"
  )
